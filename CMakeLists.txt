# Windows: every SafeHouse program gets file properties (VERSIONINFO: "SafeHouse", the program's description) and the
# manifest that makes its process code page UTF-8 (cmake/windows-utf8.manifest: argv, getenv, fopen and the narrow
# Win32 calls then take the UTF-8 paths the code passes everywhere). The games get the desktop shortcuts' icons, named
# GLFW_ICON so GLFW also uses them for the window and the taskbar.
if(WIN32)
 enable_language(RC)
 function(cane_windows_resources target description icon_lines)
  if(NOT TARGET ${target})
   return()
  endif()
  get_target_property(name ${target} OUTPUT_NAME)
  if(NOT name)
   set(name ${target})
  endif()
  set(CANE_RC_NAME "${name}")
  set(CANE_RC_DESCRIPTION "${description}")
  set(CANE_RC_MANIFEST "${CMAKE_CURRENT_SOURCE_DIR}/cmake/windows-utf8.manifest")
  set(CANE_RC_EXTRA "${icon_lines}")
  set(rc "${CMAKE_BINARY_DIR}/generated/${target}-resources.rc")
  configure_file(cmake/windows-version.rc.in "${rc}" @ONLY)
  set_source_files_properties("${rc}" PROPERTIES OBJECT_DEPENDS "${CANE_RC_MANIFEST};${ARGN}")
  target_sources(${target} PRIVATE "${rc}")
 endfunction()
